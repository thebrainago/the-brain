FLIP FULL

//@version=6
strategy("Flip Trailing Buy & Exit", overlay=true, initial_capital=10000, default_qty_type=strategy.percent_of_equity, default_qty_value=100)

// --- Thông số đầu vào ---
pct = input.float(5.0, "Tỷ lệ dời điểm dừng (%)", minval=0.1, step=0.1)

// --- Logic tính toán biến Flip (Dựa trên Trailing Stop) ---
var float flip = close

if bar_index > 0
    if flip[1] < 0
        // Trạng thái hiện tại: Chưa có tín hiệu hoặc đang chờ mua
        threshold = -flip[1] * (1 + pct / 100)
        if close > threshold
            flip := close
        else
            flip := -math.min(-flip[1], close)
    else
        // Trạng thái hiện tại: Đang trong xu hướng tăng (Long)
        threshold = flip[1] * (1 - pct / 100)
        if close < threshold
            flip := -close
        else
            flip := math.max(flip[1], close)

// --- Điều kiện vào và thoát lệnh ---
// Mua khi flip chuyển từ âm sang dương
buyCondition  = flip > 0 and flip[1] < 0
// Thoát khi flip chuyển từ dương sang âm
exitCondition = flip < 0 and flip[1] > 0

// --- Thực thi chiến lược (Chỉ Long) ---
if buyCondition
    strategy.entry("Long", strategy.long, comment="Mua")

if exitCondition
    strategy.close("Long", comment="Thoát lệnh")

// --- Hiển thị lên biểu đồ ---
// Đường trailing stop: Màu xanh khi đang giữ lệnh, Màu đỏ khi đang đứng ngoài
plotColor = flip > 0 ? color.new(color.green, 0) : color.new(color.red, 0)
plot(math.abs(flip), color=plotColor, linewidth=2, title="Trailing Stop Level")

// Đánh dấu các điểm Mua/Thoát
plotshape(buyCondition, "Buy Signal", shape.triangleup, location.belowbar, color=color.green, size=size.small)
plotshape(exitCondition, "Exit Signal", shape.triangledown, location.abovebar, color=color.red, size=size.small)
