def load_variant(name, repl=()):
    s = open(SRC_PATH, encoding="utf-8").read()
    for a, b in repl:
        assert a in s, "mau khong co trong ma: %r" % a
        s = s.replace(a, b)
    mod = types.ModuleType(name)
    mod.__file__ = SRC_PATH
    sys.modules[name] = mod
    exec(compile(s, name, "exec"), mod.__dict__)
    return mod


class Gia:
    """Thay tam mot thuoc tinh cua module, tra lai khi thoat."""
    def __init__(self, obj, **kw):
        self.obj, self.kw, self.cu = obj, kw, {}

    def __enter__(self):
        for k, v in self.kw.items():
            self.cu[k] = getattr(self.obj, k)
            setattr(self.obj, k, v)
        return self

    def __exit__(self, *a):
        for k, v in self.cu.items():
            setattr(self.obj, k, v)

def hang_gia(ten, qm, w, tam, san, hoi, flagged):
    out = []
    for o in range(3):
        r = hoi[o] if isinstance(hoi, (list, tuple)) else hoi
        out.append({"obs": o, "ten": ten, "quy_mo": qm, "w": w, "tam": tam, "san": san, "ly_do": [], "ap_duoc": True, "buoc": 20.0, "tp": 20.0,
                    "tang": 10, "diem_b": 100.0 * (1.0 - r), "hoi_tiec": r, "duoi_phan_giai": bool(flagged)})
    return out


def ket_qua_gia(M):
    """6 cap do duoc (K2, C2, H4 x phang, geo) + 2 cap Z0. Moi bien the: (quy_mo, w, tam, san, [hoi tiec 6 cap], [co duoi phan giai 6 cap])."""
    T = M.ten_bien_the
    pairs = [("K2", "L1", "phang"), ("K2", "L1", "geo"), ("C2", "L2", "phang"), ("C2", "L2", "geo"), ("H4", "L4", "phang"), ("H4", "L4", "geo")]
    F, N = True, False
    bt = {
        "B0_chep": (None, None, None, None, [0.80, 0.82, 0.30, 0.31, 0.20, 0.22], [F, F, N, N, N, N]),
        T("A_chart", 0.0, "giu", 0.0): ("A_chart", 0.0, "giu", 0.0, [0.35, 0.33, 0.12, 0.13, 0.15, 0.16], [N] * 6),
        T("A_chart", 0.0, "I6", 3.0): ("A_chart", 0.0, "I6", 3.0, [0.30, 0.28, 0.12, 0.11, 0.14, 0.15], [N] * 6),
        T("A_ref", 0.0, "giu", 0.0): ("A_ref", 0.0, "giu", 0.0, [0.10] * 6, [N] * 6),
        T("R_H", 0.5, "I6", 0.0): ("R_H", 0.5, "I6", 0.0, [0.095] * 6, [N] * 6),
        T("sigma", 1.0, "I6", 3.0): ("sigma", 1.0, "I6", 3.0, [0.05, 0.05, 0.05, 0.05, 0.05, 0.05], [N, N, N, N, F, F]),
        T("A_chart", 1.0, "giu", 0.0): ("A_chart", 1.0, "giu", 0.0, [0.14, 0.14, 0.14, 0.14, 0.0, 0.0], [N] * 6),
    }
    z0 = {"B0_chep": [0.0, 0.0], T("A_ref", 0.0, "giu", 0.0): [0.9, 0.9], T("R_H", 0.5, "I6", 0.0): [0.9, 0.9],
          T("sigma", 1.0, "I6", 3.0): [0.0, 0.0], T("A_chart", 0.0, "giu", 0.0): [0.0, 0.0], T("A_chart", 0.0, "I6", 3.0): [0.0, 0.0],
          T("A_chart", 1.0, "giu", 0.0): [0.0, 0.0]}
    res = []
    for ip, (kb, loai, kieu) in enumerate(pairs):
        hang = []
        for ten, (qm, w, tam, san, hoi, fl) in bt.items():
            hang += hang_gia(ten, qm, w, tam, san, hoi[ip], fl[ip])
        res.append({"kich_ban": kb, "loai": loai, "kieu": kieu, "k_ky_vong": None, "khung_nguon": 1, "khung_dich": 1, "nguon": {}, "dich": {},
                    "dap_an_nguon": {"buoc": 10.0, "tp": 10.0, "tang": 10}, "dap_an_dich": {"buoc": 20.0, "tp": 20.0, "tang": 10}, "ho_so_nguon": [],
                    "hang": hang})
    for iz, kieu in enumerate(("phang", "geo")):
        hang = []
        for ten, (qm, w, tam, san, hoi, fl) in bt.items():
            hang += hang_gia(ten, qm, w, tam, san, z0[ten][iz], False)
        res.append({"kich_ban": "Z0", "loai": "L0", "kieu": kieu, "k_ky_vong": None, "khung_nguon": 1, "khung_dich": 1, "nguon": {}, "dich": {},
                    "dap_an_nguon": {"buoc": 10.0, "tp": 10.0, "tang": 10}, "dap_an_dich": {"buoc": 10.0, "tp": 10.0, "tang": 10}, "ho_so_nguon": [],
                    "hang": hang})
    return res


def lenh_gia(so_chuoi_dong=29, them_chuoi_mo=True, tang_lon=3):
    """Bang lenh gia: chuoi i giu (i+1)*10 phut; (so_chuoi_dong - tang_lon) chuoi co tang toi da 1 (-> 2 tang), `tang_lon` chuoi co tang toi da 7 (-> 8);
    chuoi mo (neu co) chi co lenh tang 0 va khong dong."""
    hang = []
    t0 = pd.Timestamp("2020-01-06 00:00")
    for i in range(so_chuoi_dong):
        mo = t0 + pd.Timedelta(days=i)
        dong = mo + pd.Timedelta(minutes=(i + 1) * 10)
        tmax = 7 if i >= so_chuoi_dong - tang_lon else 1
        hang.append({"ro": i, "tang": 0, "mo": mo, "dong": dong})
        hang.append({"ro": i, "tang": tmax, "mo": mo + pd.Timedelta(minutes=1), "dong": dong})
    if them_chuoi_mo:
        mo = t0 + pd.Timedelta(days=500)
        hang.append({"ro": 9999, "tang": 0, "mo": mo, "dong": pd.NaT})
    return pd.DataFrame(hang)

def main():
    import importlib
    M = importlib.import_module("nhan.thu_chuyen")
    res = checks(M)
    n_sai = 0
    for ten, dat, ghi in res:
        print("%s  %s%s" % ("PASS" if dat else "FAIL", ten, ("   [" + ghi + "]") if ghi and not dat else ""))
        n_sai += (not dat)
    print("--- ban goc: %d phep kiem, %d hong" % (len(res), n_sai))
    ok = n_sai == 0
    song = 0
    for i, (ten, repl) in enumerate(mutants()):
        r = []
        try:
            V = load_variant("tc_mut_%d" % i, repl)
            checks(V, r)
            hong = [t for t, d, _ in r if not d]
        except Exception as e:                     # noqa: BLE001 - dot bien lam sap giua chung cung tinh la bi bat
            hong = [t for t, d, _ in r if not d] + ["(ngoai le) %s: %s" % (type(e).__name__, str(e)[:60])]
        bat = bool(hong)
        song += (not bat)
        print("%s  dot bien: %s%s" % ("BAT " if bat else "SONG", ten, ("   <- " + hong[0]) if bat else ""))
    print("--- %d dot bien, %d song sot" % (len(mutants()), song))
    return 0 if ok and song == 0 else 1
