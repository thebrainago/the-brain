def load_variant(name, repl=()):
    s = open(SRC_PATH, encoding="utf-8").read()
    for a, b in repl:
        assert a in s, "mau khong co trong ma: %r" % a
        s = s.replace(a, b, 1)
    mod = types.ModuleType(name)
    mod.__file__ = SRC_PATH
    sys.modules[name] = mod
    exec(compile(s, name, "exec"), mod.__dict__)
    return mod


class Gia:
    """Thay tam mot thuoc tinh cua module, tra lai khi thoat."""
    def __init__(self, obj, **kw):
        self.obj, self.kw, self.cu = obj, kw, {}

    def __enter__(self):
        for k, v in self.kw.items():
            self.cu[k] = getattr(self.obj, k)
            setattr(self.obj, k, v)
        return self

    def __exit__(self, *a):
        for k, v in self.cu.items():
            setattr(self.obj, k, v)

def f_that(ts):
    """Be mat diem GIA: don dinh, dinh tai (BSTAR, tp = buoc, 8 tang); dinh nam DUNG tren luoi dap an."""
    return 100.0 - 30.0 * math.log(ts.buoc / BSTAR) ** 2 - 20.0 * math.log(ts.tp / ts.buoc) ** 2 - 10.0 * math.log(ts.tran_tang / 8.0) ** 2


def ruido(ts, hat, bien_do):
    k = zlib.crc32(("%s|%d" % (DM._khoa(ts), hat)).encode("ascii")) % 10007
    return (k / 10007.0 - 0.5) * 2.0 * bien_do

UV0 = [KIEU.tham_so(31.7, 24.3, 6), KIEU.tham_so(13.3, 16.1, 11), KIEU.tham_so(60.9, 41.3, 14)]
B0 = KIEU.tham_so(18.0, 18.0, 8)
B1 = KIEU.tham_so(25.0, 25.0, 8)
SAN = 5.0


def chay_hat(M, ruido_tim=0.0, ruido_kiem=0.0, spikes=None, ung_vien=UV0, seed=1, top_k=(1, 4, 12), diem_b_da=100.0, co_be_mat=True,
             f_b=None, cong_fn=None, ts_t0="mac_dinh", san=None):
    san = SAN if san is None else san
    spikes = spikes or {}
    goi_tim, goi_b, goi_kiem = [], [], []

    def dg_tim(cells):
        goi_tim.extend(DM._khoa(c) for c in cells)
        return [f_that(c) + ruido(c, 1, ruido_tim) + spikes.get(DM._khoa(c), 0.0) for c in cells]

    def dg_kiem(cells):
        goi_kiem.append(len(cells))
        return [f_that(c) + ruido(c, 2, ruido_kiem) for c in cells]

    def dg_b(cells):
        goi_b.append(len(cells))
        return [(f_b or f_that)(c) for c in cells]

    def cong(cells):
        return [(cong_fn or (lambda c: {"dat_kham_pha": f_that(c) > 95, "dat": f_that(c) > 99, "k": 2.0, "ly_do": ""}))(c) for c in cells]

    cac, chi = T.luoi_o_dap_an(KIEU, san)
    bm = np.full((len(T.BUOC), len(T.TY_LE_TP), len(T.TANG)), np.nan)
    for (ib, ir, it), c in zip(chi, cac):
        bm[ib, ir, it] = f_that(c)
    rng = np.random.default_rng(seed)
    res = M.so_sanh_quy_trinh(KIEU, list(ung_vien), B0, UV0[0] if ts_t0 == "mac_dinh" else ts_t0, B1, san, dg_tim, dg_kiem, dg_b, cong, rng,
                              diem_b_da, bm if co_be_mat else None, top_k=top_k)
    return res, goi_tim, goi_b, goi_kiem

def hang(hoi, ap=True, kp=False, xn=False, sp=100, sk=0, dp=False):
    return {"ap_duoc": ap, "hoi_tiec": hoi, "duoi_phan_giai": dp, "so_phep_thu": sp, "so_kiem": sk, "dat_kham_pha": kp, "dat_xac_nhan": xn}


def ket_qua_gia():
    def k(kb, loai, kieu, hats):
        return {"kich_ban": kb, "loai": loai, "kieu": kieu, "hat_ket_qua": hats}

    def hat(sm, rt, rho=0.8, **ex):
        qt = {"SMART_1": hang(sm), "RANDT_1": hang(rt), "SMARTRAW_1": hang(ex.get("raw", sm)), "SMART_4": hang(sm), "RANDT_4": hang(rt),
              "SMART_S_1": hang(sm), "RANDT_S_1": hang(rt), "T0": hang(sm + 0.3), "B1": hang(sm + 0.5), "B0": hang(sm + 0.8)}
        if "bo_sm" in ex:
            qt["SMART_1"] = hang(None, ap=False)
        return {"quy_trinh": qt, "rho_xep_hang_full": rho}
    ra = []
    ra.append(k("K2", "L1", "phang", [hat(0.10, 0.20, 0.9), hat(0.12, 0.22, 0.9), hat(0.14, 0.24, 0.9), hat(0.16, 0.26, 0.9)]))
    ra.append(k("K2", "L1", "geo", [hat(0.20, 0.30, 0.8)] * 4))
    ra.append(k("C2", "L2", "phang", [hat(0.0, 0.10, 0.7)] * 4))
    ra.append(k("C2", "L2", "geo", [hat(0.30, 0.40, 0.6)] * 4))
    ra.append(k("Z0", "L0", "phang", [hat(0.05, 0.05, 1.0)] * 4))
    nh = []
    for i in range(4):
        q = {"SMART_1": hang(None, kp=(i < 3), xn=(i == 0)), "RANDT_1": hang(None, kp=True, xn=False)}
        nh.append({"quy_trinh": q, "rho_xep_hang_full": None})
    ra.append(k("N0", "L5", "phang", nh))
    return ra

def main():
    import importlib
    M = importlib.import_module("nhan.thu_chuyen_tim")
    res = checks(M)
    n_sai = 0
    for ten, dat, ghi in res:
        print("%s  %s%s" % ("PASS" if dat else "FAIL", ten, ("   [" + ghi + "]") if ghi and not dat else ""))
        n_sai += (not dat)
    print("--- ban goc: %d phep kiem, %d hong" % (len(res), n_sai))
    ok = n_sai == 0
    song = 0
    nen_hong = set(t for t, d, _ in res if not d)
    for i, entry in enumerate(mutants()):
        ten, repl = entry[0], entry[1]
        thuc = len(entry) > 2 and entry[2] == "thuc"       # dot bien chi nam trong duong ong that (giai_doan_2): chay kem phan F
        if os.environ.get("KIEM_CHI") == "thuc" and not thuc:    # chay nhanh: chi cac dot bien cua duong ong that
            continue
        r = []
        try:
            V = load_variant("tct_mut_%d" % i, repl)
            checks(V, r, day_du=thuc)
            hong = [t for t, d, _ in r if not d and t not in nen_hong]
        except Exception as e:                     # noqa: BLE001 - dot bien lam sap giua chung cung tinh la bi bat
            hong = [t for t, d, _ in r if not d and t not in nen_hong] + ["(ngoai le) %s: %s" % (type(e).__name__, str(e)[:60])]
        bat = bool(hong)
        song += (not bat)
        print("%s  dot bien: %s%s" % ("BAT " if bat else "SONG", ten, ("   <- " + hong[0]) if bat else ""))
    print("--- %d dot bien, %d song sot" % (len(mutants()), song))
    return 0 if ok and song == 0 else 1


if __name__ == "__main__":
    sys.exit(main())
