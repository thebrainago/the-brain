def load_variant(name, repl=()):
    s = open(SRC_PATH, encoding="utf-8").read()
    for a, b in repl:
        assert a in s, "mau khong co trong ma: %r" % a
        s = s.replace(a, b)
    mod = types.ModuleType(name)
    mod.__file__ = SRC_PATH
    sys.modules[name] = mod
    exec(compile(s, name, "exec"), mod.__dict__)
    return mod

def paths(seed=1, n15=60000, s1=5e-5, base=1.0):
    """Duong gia ngau nhien phut -> nen M15 va H1 CUNG MOT duong (cho I6 theo gio dong ho)."""
    rng = np.random.default_rng(seed)
    cl1 = base + np.cumsum(rng.normal(0.0, s1, n15 * 15))
    out = {}
    for ten, k, fr in (("m15", 15, "15min"), ("h1", 60, "60min")):
        n = len(cl1) // k
        w = cl1[: n * k].reshape(n, k)
        idx = pd.date_range("2020-01-01", periods=n, freq=fr)
        out[ten] = pd.DataFrame({"high": w.max(1), "low": w.min(1), "close": w[:, -1], "spread": 20.0}, index=idx)
    return out

def tt_tay(D, **kw):
    base = dict(ma="X", khung_phut=15.0, A=20e-4, C=2.5e-4, pip=1e-4, pv=1e5, von=10000.0, do_tin_chi_phi="DO")
    base.update(kw)
    return D.ThiTruong(**base)


def xuong(P0, tam, chieu, bars=60):
    """Duong gia tu P0 di NGUOC chieu lenh `chieu` dung `tam` (don vi gia); thanh cuoi cham moc cuoi (+1e-9 tuong doi)."""
    cl = P0 - chieu * np.linspace(0.0, tam, bars)
    prev = np.r_[cl[0], cl[:-1]]
    hi = np.maximum(prev, cl)
    lo = np.minimum(prev, cl)
    if chieu > 0:
        lo[-1] = cl[-1] * (1.0 - 1e-9)
    else:
        hi[-1] = cl[-1] * (1.0 + 1e-9)
    idx = pd.date_range("2020-01-01", periods=bars, freq="15min")
    return pd.DataFrame({"high": hi, "low": lo, "close": cl}, index=idx)


def chay_xuong(D, ts, qc, P0, von, n):
    """Chay engine that tren duong gia xuong den tang thu n; tra (tang_max, lo_treo_dinh, du doan_cua_module)."""
    hr = D.hinh_rui_ro(ts.buoc, ts.he_so_buoc, ts.buoc_tran, ts.kieu_lot, ts.he_so_lot, n)
    chieu = 1 if ts.che_do == "mua" else -1
    df = xuong(P0, hr["tam"] * qc.pip, chieu)
    ts2 = dataclasses.replace(ts, tran_tang=n, tp=100000.0)
    kq = LU.chay_mang(LU.chuan_bi(df, qc), ts2, von)
    du_doan = ts.lot * qc.hop_dong * hr["mat_tai_tam"] * qc.pip
    return kq.tang_max, kq.lo_treo_dinh, du_doan

        # so nen theo khung bieu do: chi doi khi bat co va khung khac nhau
        h1 = dataclasses.replace(audc, khung_phut=60.0)
        kh1 = D.dich_luoi(tsc, gold, h1, D.CachDich(), None, tang_tham_chieu=12)
        r0 = D.dich_bo_set(sets["cancubo"], kh1, gold, h1)
        r1 = D.dich_bo_set(sets["cancubo"], kh1, gold, h1, nen_theo_khung_bieu_do=True)
        ck("dich_set_nen_mac_dinh_giu", "InpEMA1" not in r0["doi"] and "InpEMA1" in r1["doi"] and int(r1["doi"]["InpEMA1"]) == max(1, int(round(34 * 15 / 60.0))),
           (r0["doi"].get("InpEMA1"), r1["doi"].get("InpEMA1")))
    else:
        ck("dich_set_phan_hoach", False, kc.ly_do)

    # 15. nhieu cach dich: tran 9, gop trung, khong mat cach nao
    cac = D.cac_cach_dich(tsc, gold, audc, None, 12)
    tong = sum(1 + len(k.cung_ket_qua) for k in cac)
    ck("cac_cach_dich_gop_trung", 1 <= len(cac) <= D.TOI_DA_CACH_DICH and tong in (8, 9) and len({dataclasses.astuple(k.tham_so) for k in cac if k.tham_so is not None}) == len([k for k in cac if k.tham_so is not None]),
       (len(cac), tong))
    audx = D.ThiTruong(ma="AUDCAD", khung_phut=15, A=6e-4, C=2.5e-4 * 6, pip=1e-4, pv=73000.0, von=10000.0, do_tin_chi_phi="DO", tien_te="tai_khoan", point=1e-5)
    cac2 = D.cac_cach_dich(tsc, gold, audx, None, 12)
    ck("them_I2_khi_chi_phi_lech", sum(1 + len(k.cung_ket_qua) for k in cac2) == 9 and any(k.cach.w_buoc == 1.0 for k in cac2), [k.cach.ten() for k in cac2])
    cac3 = D.cac_cach_dich(tsc, gold, audx, None, 12, lot="giu")
    ck("cac_cach_dich_lot_tham_so", len(cac3) >= 1 and all(k.cach.lot == "giu" for k in cac3) and all(k.cach.lot == "I4" for k in cac2)
       and any(k.cach.w_buoc == 1.0 for k in cac3), ([k.cach.lot for k in cac3], [k.cach.lot for k in cac2]))

    # 16. I6: gio dong ho vs so nen (M15 -> H1 CUNG duong gia)
    if not nhanh:
        P = paths(seed=5, n15=60000)
        t15 = D.ThiTruong.tu_du_lieu(P["m15"], QC, 1e6)
        t60 = D.ThiTruong.tu_du_lieu(P["h1"], QC, 1e6)
        tsi = LU.ThamSo(buoc=60.0, tp=60.0, tran_tang=40, che_do="mua", lot=1.0)
        kd = D.dich_luoi(tsi, t15, t60, D.CachDich(w_buoc=0.0, tam="I6", thoi_gian="dong_ho"), 240.0, 12)
        kn = D.dich_luoi(tsi, t15, t60, D.CachDich(w_buoc=0.0, tam="I6", thoi_gian="so_nen"), 240.0, 12)
        if kd.trang_thai == D.OK and kn.trang_thai == D.OK:
            ktd = [c for c in kd.chi_tiet if c["ten"] == "tam"][0]["he_so"]
            ktn = [c for c in kn.chi_tiet if c["ten"] == "tam"][0]["he_so"]
            ck("I6_dong_ho_vs_so_nen", ktd is not None and ktn is not None and 0.85 < ktd < 1.2 and 1.7 < ktn < 2.3 and kd.tham_so.tran_tang < kn.tham_so.tran_tang,
               (ktd, ktn, kd.tham_so.tran_tang, kn.tham_so.tran_tang))
        else:
            ck("I6_dong_ho_vs_so_nen", False, (kd.ly_do, kn.ly_do))
        # khong co gia_giu -> canh bao, giu so tang
        kg = D.dich_luoi(tsi, t15, t60, D.CachDich(w_buoc=0.0, tam="I6"), None, 12)
        ck("I6_thieu_gio_giu_canh_bao", kg.trang_thai == D.OK and any("gio_giu_phut" in x for x in kg.canh_bao))

    # 17. dau ra dung duoc: JSON + tom tat
    try:
        j1 = json.dumps(kc.to_dict())
        j2 = json.dumps(D.dich_luoi(ts0, A, Bv).to_dict())
        ok_json = len(j1) > 100 and len(j2) > 50
    except Exception as e:      # noqa: BLE001
        ok_json = False
        j1 = repr(e)
    ck("to_dict_json", ok_json, j1[:100] if not ok_json else "")
    tt1 = kc.tom_tat() if kc.trang_thai == D.OK else []
    tt2 = D.dich_luoi(ts0, A, Bv).tom_tat()
    ck("tom_tat_ngan", 1 <= len(tt1) <= 8 and 1 <= len(tt2) <= 8 and all(isinstance(x, str) for x in tt1 + tt2), (len(tt1), len(tt2)))

    # 18. canh bao rui ro khi tam > tran (mat >= 80% von) va canh bao chi phi KHAI
    tsx = LU.ThamSo(buoc=100.0, tp=100.0, tran_tang=14, che_do="mua", lot=4.0, kieu_lot="nhan", he_so_lot=1.3)
    kx = D.dich_luoi(tsx, vang, vang if False else tt_tay(D, ma="Y", A=2.0, C=0.17, pip=0.1, pv=100.0, von=1e6, point=0.01, do_tin_chi_phi="KHAI"), D.CachDich(tam="giu"))
    ck("canh_bao_KHAI", kx.trang_thai == D.OK and any("KHAI" in x for x in kx.canh_bao), kx.canh_bao)
    ky = D.dich_luoi(tsx, vang, tt_tay(D, ma="Y", A=2.0, C=0.17, pip=0.1, pv=100.0, von=1e6 / 40.0, point=0.01, do_tin_chi_phi="DO"), D.CachDich(tam="giu", lot="giu"))
    ck("canh_bao_rui_ro_tran", ky.trang_thai == D.OK and any("tran sut giam" in x for x in ky.canh_bao), ky.canh_bao)
    return R


def chay(D, ten, nhanh=False):
    R = checks(D, nhanh=nhanh)
    hong = [k for k, (ok, _) in R.items() if not ok]
    return R, hong
